2Prisma স্কিমা (Auth.js মডেলসহ)

prisma/schema.prisma
generator client { provider = "prisma-client-js" }
datasource db { provider = "mongodb"; url = env("DATABASE_URL") }

model User {
  id            String    @id @default(auto()) @map("_id") @db.ObjectId
  name          String?
  email         String?   @unique
  emailVerified DateTime?
  image         String?
  accounts      Account[]
  posts         Post[]
}
model Account {
  id                String  @id @default(auto()) @map("_id") @db.ObjectId
  userId            String  @db.ObjectId
  type              String
  provider          String
  providerAccountId String
  refresh_token     String? @db.String
  access_token      String? @db.String
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String? @db.String
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
  @@unique([provider, providerAccountId])
}
model Post {
  id       String @id @default(auto()) @map("_id") @db.ObjectId
  title    String
  authorId String @db.ObjectId
  author   User   @relation(fields: [authorId], references: [id])
}
npx prisma generate
npx prisma db push   # MongoDB-তে migrate নয়, db push
💡 JWT সেশন ব্যবহার করলে Session/VerificationToken মডেল লাগে না। Email লিংক লগইন চাইলে VerificationToken যোগ করতে হবে।

🔍 লাইনে লাইনে ব্যাখ্যা — কেন এটা ব্যবহার হলো

@map("_id") @db.ObjectIdMongoDB-র আইডি হলো _id ObjectId, Prisma-কে সেটা জানাতে হয়
@unique (email)একই ইমেইলে দুই ইউজার ঠেকায়
@@unique([provider, providerAccountId])একই Google অ্যাকাউন্ট দুইবার জমা হয় না
onDelete: Cascadeইউজার মুছলে তার Account-ও মুছে যায়, এলোমেলো ডেটা থাকে না
prisma db pushMongoDB-তে স্কিমা সরাসরি প্রয়োগ হয়, migrate লাগে না

✅ globalThis ট্রিক না দিলে dev মোডে হট রিলোডে বারবার নতুন কানেকশন খুলবে।
src/app/dashboard/actions.ts
"use server";
import { prisma } from "@/lib/prisma";
import { auth } from "@/auth";
import { revalidatePath } from "next/cache";

export async function addPost(fd: FormData) {
  const session = await auth();
  if (!session?.user?.id) throw new Error("লগইন দরকার");
  await prisma.post.create({ data: { title: String(fd.get("title")), authorId: session.user.id } });
  revalidatePath("/dashboard");
}

5Auth.js কনফিগ

src/auth.ts
import NextAuth from "next-auth";
import Google from "next-auth/providers/google";
import { PrismaAdapter } from "@auth/prisma-adapter";
import { prisma } from "@/lib/prisma";

export const { handlers, auth, signIn, signOut } = NextAuth({
  adapter: PrismaAdapter(prisma),
  providers: [Google],
  session: { strategy: "jwt" },
  callbacks: {
    jwt({ token, user }) { if (user) token.id = user.id; return token; },
    session({ session, token }) { if (token.id) session.user.id = token.id as string; return session; },
  },
});
src/app/api/auth/[...nextauth]/route.ts
import { handlers } from "@/auth";
export const { GET, POST } = handlers;
💡 session.user.id TypeScript-এ এরর দিলে next-auth.d.ts-এ Session টাইপে id: string যোগ করো।

🔍 লাইনে লাইনে ব্যাখ্যা — কেন এটা ব্যবহার হলো

PrismaAdapter(prisma)গুগলে প্রথমবার ঢুকলে User ও Account নিজে ডাটাবেজে বানায়
providers: [Google]AUTH_GOOGLE_ID/SECRET env থেকে নিজেই পড়ে
strategy: "jwt"সেশন DB-তে নয়, কুকির টোকেনে থাকে; দ্রুত ও সহজ
callbacks.jwtলগইনের সময় user.id টোকেনে তোলে
callbacks.sessionটোকেন থেকে id সেশনে আনে, তাই auth() দিয়ে session.user.id মেলে
handlers → GET, POST/api/auth/* এর সব রুট (signin, callback) এই এক লাইনে চালু

6লগইন বাটন ও প্রটেক্টেড পেজ

src/components/AuthButton.tsx
import { signIn, signOut, auth } from "@/auth";

export default async function AuthButton() {
  const session = await auth();
  return session ? (
    <form action={async () => { "use server"; await signOut({ redirectTo: "/" }); }}>
      <span>{session.user?.name}</span> <button>লগআউট</button>
    </form>
  ) : (
    <form action={async () => { "use server"; await signIn("google", { redirectTo: "/dashboard" }); }}>
      <button>Google দিয়ে লগইন</button>
    </form>
  );
}
src/app/dashboard/page.tsx
import { auth } from "@/auth";
import { redirect } from "next/navigation";
import { prisma } from "@/lib/prisma";

export default async function Dashboard() {
  const session = await auth();
  if (!session?.user?.id) redirect("/");
  const posts = await prisma.post.findMany({ where: { authorId: session.user.id } });
  return <h1>স্বাগতম {session.user.name} — তোমার পোস্ট: {posts.length}</h1>;
}
