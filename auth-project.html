2Session লাইব্রেরি

src/lib/session.ts
import "server-only";
import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";

const key = new TextEncoder().encode(process.env.SESSION_SECRET);

export async function createSession(userId: string) {
  const token = await new SignJWT({ userId })
    .setProtectedHeader({ alg: "HS256" })
    .setExpirationTime("7d").sign(key);
  (await cookies()).set("session", token, { httpOnly: true, secure: true, sameSite: "lax", path: "/" });
}

export async function getSession() {
  const token = (await cookies()).get("session")?.value;
  if (!token) return null;
  try { return (await jwtVerify(token, key)).payload as { userId: string }; }
  catch { return null; }
}

export async function deleteSession() { (await cookies()).delete("session"); }
⚠️ httpOnly কুকি JavaScript থেকে পড়া যায় না — XSS থেকে টোকেন বাঁচে। localStorage-এ টোকেন রেখো না।

🔍 লাইনে লাইনে ব্যাখ্যা — কেন এটা ব্যবহার হলো

import "server-only"ভুল করে Client Component-এ ইমপোর্ট হলে বিল্ড এরর দেয়, তাই SECRET ব্রাউজারে যেতে পারে না
new TextEncoder().encodejose-র কী Uint8Array চায়, স্ট্রিং থেকে বদলায়
SignJWT(...).sign(key)payload-এ সিল বসিয়ে টোকেন বানায়
httpOnly / secure / sameSiteকুকি সুরক্ষার তিন ফ্ল্যাগ (বিস্তারিত JWT পেজে)
await cookies()Next 15+ এ cookies() Promise ফেরত দেয়
try { jwtVerify } catch → nullমেয়াদোত্তীর্ণ বা জাল টোকেনে ক্র্যাশ না করে "লগইন নেই" ধরা হয়

3Signup ও Login (Server Action)

src/app/actions.ts
"use server";
import bcrypt from "bcryptjs";
import { redirect } from "next/navigation";
import { createSession, deleteSession } from "@/lib/session";

const users: { id: string; email: string; hash: string }[] = []; // ডেমো: আসলে DB ব্যবহার করো

export async function signup(_: unknown, fd: FormData) {
  const email = String(fd.get("email")), pass = String(fd.get("password"));
  if (pass.length < 6) return { error: "পাসওয়ার্ড কমপক্ষে ৬ অক্ষর" };
  const user = { id: crypto.randomUUID(), email, hash: await bcrypt.hash(pass, 10) };
  users.push(user);
  await createSession(user.id);
  redirect("/dashboard");
}

export async function login(_: unknown, fd: FormData) {
  const u = users.find(x => x.email === fd.get("email"));
  if (!u || !(await bcrypt.compare(String(fd.get("password")), u.hash)))
    return { error: "ইমেইল বা পাসওয়ার্ড ভুল" };
  await createSession(u.id);
  redirect("/dashboard");
}

export async function logout() { await deleteSession(); redirect("/login"); }

4রুট প্রটেকশন

src/proxy.ts (Next 15 হলে middleware.ts, ফাংশনের নাম middleware)
import { NextRequest, NextResponse } from "next/server";
import { jwtVerify } from "jose";

const key = new TextEncoder().encode(process.env.SESSION_SECRET);

export default async function proxy(req: NextRequest) {   // Next 16: proxy.ts, আগে: middleware.ts
  const token = req.cookies.get("session")?.value;
  try { if (token) await jwtVerify(token, key); else throw 0; }
  catch { return NextResponse.redirect(new URL("/login", req.url)); }
}
export const config = { matcher: ["/dashboard/:path*"] };
src/app/dashboard/page.tsx
import { getSession } from "@/lib/session";
import { logout } from "../actions";
import { redirect } from "next/navigation";

export default async function Dashboard() {
  const s = await getSession();
  if (!s) redirect("/login");          // দ্বিতীয় নিরাপত্তা স্তর
  return <form action={logout}><h1>স্বাগতম! ID: {s.userId}</h1><button>লগআউট</button></form>;
}
